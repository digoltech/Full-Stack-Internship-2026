## Section E — Debugging Challenge

Find and fix the problems in this component.

```jsx
import { useState } from "react";

function LoginForm() {
  const [email, setEmail] = useState("");

  function handleSubmit(event) {
    console.log("Email:", email);
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="email"
        value={email}
        onChange={(event) => setEmail(event.target.value)}
      />

      <button>Login</button>
    </form>
  );
}

export default LoginForm;
```

### Questions

1. Why might the page reload or navigate during submission?
2. What is missing from the input for better accessibility?
3. How can we make sure the email is not empty?
4. Why should a production login form avoid logging credentials?

### One Possible Correction

```jsx
import { useState } from "react";

function LoginForm() {
  const [email, setEmail] = useState("");
  const [error, setError] = useState("");

  function handleSubmit(event) {
    event.preventDefault();

    if (!email.trim()) {
      setError("Email is required.");
      return;
    }

    setError("");
    alert("Email entered successfully.");
  }

  return (
    <form onSubmit={handleSubmit}>
      <label htmlFor="email">Email</label>

      <input
        id="email"
        name="email"
        type="email"
        value={email}
        onChange={(event) => setEmail(event.target.value)}
      />

      {error && <p>{error}</p>}

      <button type="submit">Continue</button>
    </form>
  );
}

export default LoginForm;
```
