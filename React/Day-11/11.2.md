### Section B — Expected Points

1. Form submit → validate → `POST /auth/login` → server verifies → success response with token and user → app stores session (state and storage) → navigate to the requested page → protected route sees `isAuthenticated` and renders the page → requests send `Authorization: Bearer <token>`.
2. `localStorage` can be read by any JavaScript running on the page, including injected scripts (XSS), so a stolen token is possible. HttpOnly cookies cannot be read by JavaScript, so they are safer against XSS (but need CSRF protection and a real backend).
3. It redirects with `<Navigate to="/login" state={{ from: location }} replace />`, and the login page reads `location.state.from` to navigate back after login.
4. Any four of: visible `<label>`s, `aria-invalid` / `aria-describedby` for errors, `role="alert"` for server errors, keyboard focus order, `autocomplete` attributes, sufficient colour contrast, disabled button while submitting, `type="password"`.
5. The dev server hides problems: case-sensitive paths, missing env variables, production-only errors. Preview tests the real production bundle.

### Section C

### Section E — Problems and Fix

Problems:

1. `createContext()` without a guard; the consumers use `useContext` directly and `useAuth` is never defined.
2. `JSON.parse(localStorage.getItem("user"))` runs on every render and can throw; use a lazy initialiser with `try/catch`.
3. Login request has no `Content-Type: application/json` header.
4. No `res.ok` check; failed logins would be stored as a "user".
5. **The password is stored in `localStorage`.** Severe security issue. Remove it. Store only the token and user information.
6. Raw login response (including `refreshToken`) is stored without need.
7. `login` errors are not handled and not thrown; the form cannot show an error.
8. No `logout`, no token exposed, no `isAuthenticated`, no session check state.
9. `ProtectedRoute` assigns `window.location.href` during render: full-page reload, side effect in render, and the component still returns `children`. Use `<Navigate>`.
10. The protected route does not remember the target page.
11. `LoginPage` uses `useAuth` which is not defined and `navigate` which is not imported.
12. `handleSubmit` does not call `event.preventDefault()`, does not `await` the login, and navigates even if the login fails.
13. Inputs have no labels, the password input has no `type="password"`, there is no error message and no `disabled` state.
14. The context value is recreated on every render (minor).

Corrected version (condensed):

```jsx
// AuthContext.jsx
const AuthContext = createContext(null);

function loadSession() {
  try {
    return JSON.parse(localStorage.getItem("session")) ?? null;
  } catch {
    return null;
  }
}

export function AuthProvider({ children }) {
  const [session, setSession] = useState(loadSession);

  useEffect(() => {
    if (session) localStorage.setItem("session", JSON.stringify(session));
    else localStorage.removeItem("session");
  }, [session]);

  const login = useCallback(async (username, password) => {
    const res = await fetch("https://dummyjson.com/auth/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ username, password })
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) throw new Error(data.message || "Login failed");

    const { accessToken, refreshToken, ...user } = data;
    setSession({ user, token: accessToken });
  }, []);

  const logout = useCallback(() => setSession(null), []);

  const value = useMemo(
    () => ({ user: session?.user ?? null, token: session?.token ?? null, isAuthenticated: Boolean(session), login, logout }),
    [session, login, logout]
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}
```

```jsx
// LoginPage.jsx
function LoginPage() {
  const { login } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();
  const redirectTo = location.state?.from?.pathname ?? "/dashboard";

  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [submitting, setSubmitting] = useState(false);

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");
    try {
      setSubmitting(true);
      await login(username.trim(), password);
      navigate(redirectTo, { replace: true });
    } catch (err) {
      setError(err.message);
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <label htmlFor="username">Username</label>
      <input id="username" value={username} onChange={(e) => setUsername(e.target.value)} autoComplete="username" />

      <label htmlFor="password">Password</label>
      <input id="password" type="password" value={password} onChange={(e) => setPassword(e.target.value)} autoComplete="current-password" />

      {error && <p role="alert">{error}</p>}
      <button type="submit" disabled={submitting}>{submitting ? "Logging in..." : "Login"}</button>
    </form>
  );
}
```
