### Section B — Expected Points

1. A single-page application loads one HTML page and JavaScript, then swaps components on URL changes without a full reload. A multi-page site requests a new HTML document from the server on each navigation.
2. A route without a path that renders a component containing `<Outlet />`. Child routes render inside the `Outlet`, so shared parts (navbar, footer) are written once.
3. `Link` is a clickable element in JSX. `useNavigate` navigates from code, for example after a successful form submit or login.
4. When the URL changes from `/products/1` to `/products/2`, the same component stays mounted. Without the dependency the effect would not re-run, showing stale data.
5. Params identify a resource (`/products/5`). Query parameters describe how to view a list (search, sort, filter, page).

```jsx
// ProductDetails.jsx
import { useEffect, useState } from "react";
import { useParams } from "react-router-dom";

function ProductDetails() {
  const { id } = useParams();
  const [product, setProduct] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    const controller = new AbortController();

    async function load() {
      try {
        setLoading(true);
        setError("");
        const res = await fetch(`https://dummyjson.com/products/${id}`, {
          signal: controller.signal
        });
        if (!res.ok) throw new Error(`Status ${res.status}`);
        setProduct(await res.json());
      } catch (err) {
        if (err.name !== "AbortError") setError(err.message);
      } finally {
        if (!controller.signal.aborted) setLoading(false);
      }
    }

    load();
    return () => controller.abort();
  }, [id]);

  if (loading) return <p>Loading...</p>;
  if (error) return <p role="alert">{error}</p>;

  return <h1>{product.title}</h1>;
}
```
